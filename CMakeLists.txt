cmake_minimum_required(VERSION 3.24)
project(FeatherCast VERSION 0.1.0 LANGUAGES C CXX)
option(FEATHERCAST_BUILD_APP "Build Qt desktop application" ON)
option(FEATHERCAST_BUILD_ENGINE "Build Windows libobs backend" ON)
include(CTest)
add_subdirectory(src/core)

if(FEATHERCAST_BUILD_ENGINE OR FEATHERCAST_BUILD_APP)
    if(NOT WIN32)
        message(FATAL_ERROR "Backend targets Windows. Use -DFEATHERCAST_BUILD_ENGINE=OFF -DFEATHERCAST_BUILD_APP=OFF for portable core.")
    endif()
    set(_manifest "${CMAKE_SOURCE_DIR}/work/deps/manifest.json")
    if(EXISTS "${_manifest}")
        file(READ "${_manifest}" _deps)
        foreach(pair "OBS_SOURCE;obs_source" "OBS_SDK;obs_sdk" "OBS_RUNTIME;obs_runtime" "FFMPEG_HEADERS;ffmpeg_headers_path")
            list(GET pair 0 _key)
            list(GET pair 1 _field)
            string(JSON _value GET "${_deps}" "${_field}")
            set(FEATHERCAST_${_key} "${_value}" CACHE PATH "FeatherCast dependency ${_key}")
        endforeach()
        string(JSON _qt GET "${_deps}" qt_prefix)
        list(PREPEND CMAKE_PREFIX_PATH "${_qt}")
    endif()
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
    # FFmpeg source trees contain a file named `version`, which shadows the C++
    # standard <version> header. Expose only installed-style header directories.
    set(_ffmpeg_include "${CMAKE_BINARY_DIR}/ffmpeg-include")
    foreach(component libavformat libavcodec libavutil)
        file(COPY "${FEATHERCAST_FFMPEG_HEADERS}/${component}" DESTINATION "${_ffmpeg_include}" FILES_MATCHING PATTERN "*.h")
    endforeach()
    foreach(lib obs avformat avcodec avutil)
        find_library(FC_${lib}_LIBRARY NAMES ${lib} PATHS "${FEATHERCAST_OBS_SDK}/lib" REQUIRED NO_DEFAULT_PATH)
    endforeach()
    add_library(feathercast_engine STATIC
        src/engine/engine.cpp src/engine/relay_output.cpp
        include/feathercast/engine/engine.hpp)
    set_target_properties(feathercast_engine PROPERTIES AUTOMOC ON)
    target_compile_features(feathercast_engine PUBLIC cxx_std_20)
    target_compile_definitions(feathercast_engine PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    file(TO_CMAKE_PATH "${FEATHERCAST_OBS_RUNTIME}" _dev_runtime)
    target_compile_definitions(feathercast_engine PRIVATE FEATHERCAST_DEV_RUNTIME="${_dev_runtime}")
    target_include_directories(feathercast_engine PUBLIC include)
    target_include_directories(feathercast_engine SYSTEM PRIVATE
        "${FEATHERCAST_OBS_SOURCE}/libobs" "${FEATHERCAST_OBS_SDK}/include" "${_ffmpeg_include}")
    target_link_libraries(feathercast_engine PUBLIC feathercast_core Qt6::Core PRIVATE
        "${FC_obs_LIBRARY}" "${FC_avformat_LIBRARY}" "${FC_avcodec_LIBRARY}" "${FC_avutil_LIBRARY}" crypt32 psapi)
    if(MINGW)
        target_compile_options(feathercast_engine PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    function(feathercast_runtime target)
        foreach(helper obs-ffmpeg-mux.exe obs-nvenc-test.exe obs-qsv-test.exe)
            if(EXISTS "${FEATHERCAST_OBS_RUNTIME}/bin/64bit/${helper}")
                add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${FEATHERCAST_OBS_RUNTIME}/bin/64bit/${helper}" "$<TARGET_FILE_DIR:${target}>/${helper}")
            endif()
        endforeach()
    endfunction()
    if(BUILD_TESTING)
        add_executable(feathercast_credential_storage_tests tests/credential_storage_tests.cpp)
        target_link_libraries(feathercast_credential_storage_tests PRIVATE feathercast_engine)
        add_test(NAME feathercast_credential_storage_tests COMMAND feathercast_credential_storage_tests)
        add_executable(feathercast_engine_smoke tests/engine_smoke.cpp)
        target_link_libraries(feathercast_engine_smoke PRIVATE feathercast_engine)
        target_include_directories(feathercast_engine_smoke SYSTEM PRIVATE "${FEATHERCAST_OBS_SOURCE}/libobs" "${FEATHERCAST_OBS_SDK}/include")
        feathercast_runtime(feathercast_engine_smoke)
        # Hardware/desktop-dependent smoke is run explicitly, not on headless CI.
    endif()
    add_executable(feathercast_remux src/engine/remux_main.cpp)
    target_compile_features(feathercast_remux PRIVATE cxx_std_20)
    target_include_directories(feathercast_remux SYSTEM PRIVATE "${_ffmpeg_include}")
    target_link_libraries(feathercast_remux PRIVATE Qt6::Core "${FC_avformat_LIBRARY}" "${FC_avcodec_LIBRARY}" "${FC_avutil_LIBRARY}")
    if(FEATHERCAST_BUILD_APP AND EXISTS "${CMAKE_SOURCE_DIR}/src/app/main.cpp")
        file(GLOB_RECURSE FC_APP_SOURCES CONFIGURE_DEPENDS src/app/*.cpp src/ui/*.cpp src/ui/*.hpp include/feathercast/ui/*.hpp)
        add_executable(FeatherCast WIN32 ${FC_APP_SOURCES})
        set_target_properties(FeatherCast PROPERTIES AUTOMOC ON)
        target_link_libraries(FeatherCast PRIVATE feathercast_engine Qt6::Widgets Qt6::Gui)
        feathercast_runtime(FeatherCast)
        add_dependencies(FeatherCast feathercast_remux)
        if(BUILD_TESTING)
            add_executable(feathercast_automation_config_tests tests/automation_config_tests.cpp src/ui/automation_config.cpp)
            target_link_libraries(feathercast_automation_config_tests PRIVATE feathercast_core Qt6::Widgets)
            add_test(NAME feathercast_automation_config_tests COMMAND feathercast_automation_config_tests)
        endif()
    endif()
endif()
