add_library(feathercast_core STATIC delay_buffer.cpp automation.cpp module_registry.cpp)
target_compile_features(feathercast_core PUBLIC cxx_std_20)
target_include_directories(feathercast_core PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../../include)
if(MSVC)
    target_compile_options(feathercast_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(feathercast_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(feathercast_core_tests ../../tests/core_tests.cpp ../../examples/delay_preset_plugin.cpp)
    target_link_libraries(feathercast_core_tests PRIVATE feathercast_core)
    add_test(NAME feathercast_core_tests COMMAND feathercast_core_tests)
endif()
